cmake_minimum_required(VERSION 3.16)
project(WebmaniaLua VERSION 0.1.0 LANGUAGES C)
find_package(Webmania CONFIG REQUIRED)
find_package(Lua 5.4 EXACT REQUIRED)
find_package(Threads REQUIRED)
if(NOT LUA_VERSION_STRING MATCHES "^5\\.4\\.")
  message(FATAL_ERROR "Lua 5.4 is required")
endif()
add_library(webmania_lua MODULE src/runtime.c src/generated.c)
set_target_properties(webmania_lua PROPERTIES PREFIX "" OUTPUT_NAME "webmania" C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_compile_definitions(webmania_lua PRIVATE _POSIX_C_SOURCE=200809L)
target_include_directories(webmania_lua PRIVATE ${LUA_INCLUDE_DIR})
target_link_libraries(webmania_lua PRIVATE Webmania::webmania Threads::Threads)
target_compile_options(webmania_lua PRIVATE -Wall -Wextra -Wpedantic -Werror)
set(LUA_LIBDIR "lib/lua/5.4" CACHE PATH "Lua module install directory")
set(LUA_LUADIR "share/lua/5.4" CACHE PATH "Lua source install directory")
install(TARGETS webmania_lua LIBRARY DESTINATION ${LUA_LIBDIR})
install(FILES lua/webmania_annotations.lua lua/webmania_helpers.lua DESTINATION ${LUA_LUADIR})
install(FILES LICENSE README.md dependency-lock.json metadata/api-metadata.json DESTINATION share/webmania-lua)
